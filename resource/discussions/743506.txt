```python
# Displayed scores expressed in hundredths.
observed = {0, 1, 3, 5, 6, 8, 10, 12, 13}

def compatible(n):
    possible = {100 * k // n for k in range(n + 1)}
    return observed <= possible

print([n for n in range(1, 79) if compatible(n)])
# [58]
```
