# CELL 0 markdown
# Gemma 4 Developer Agent: Advanced EDA

This competition is **SWE-bench-style**: a declarative Gemma-4 ADK agent gets a repo snapshot plus a problem statement and must produce a patch.
The score is the **resolution rate**, the share of tasks where the hidden `test_patch` passes under `pytest`.

This notebook profiles every artifact the harness uses (tasks, gold patches, test patches, code graphs, embeddings, repo snapshots, wheels) and turns them into **agent-design decisions**.

**Sections**
1. Data inventory
2. Tasks: repos, timeline, duplicates
3. Problem statements: length, token cost, signal vs noise
4. Gold-patch anatomy: files, hunks, churn, new files
5. Test-patch anatomy: test targets, test functions
6. Localization difficulty: does the issue name the code to change?
7. Difficulty model: correlations and tiers
8. Code-intelligence tools: graph and embedding coverage
9. Repo snapshots: streamed scan, touched-file sizes vs the `read_file` budget
10. Wheels and sandbox environment
11. Key findings

# CELL 11 code
HUNK_RE = re.compile(r'^@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@ ?(.*)$')

def parse_patch(text):
    # count-aware unified-diff parser, so '---' content lines are never mistaken for headers
    files, cur, lines, i = [], None, text.splitlines(), 0
    old_hdr = None
    while i < len(lines):
        ln = lines[i]
        if ln.startswith('--- ') and i + 1 < len(lines) and lines[i+1].startswith('+++ '):
            old_hdr, new_hdr = ln[4:].split('\t')[0].strip(), lines[i+1][4:].split('\t')[0].strip()
            path = new_hdr if new_hdr != '/dev/null' else old_hdr
            path = re.sub(r'^[ab]/', '', path)
            cur = dict(path=path, add=0, dele=0, hunks=0, is_new=old_hdr == '/dev/null', is_deleted=new_hdr == '/dev/null',
                       ctx_symbols=[], added_defs=[], ranges=[])
            files.append(cur); i += 2; continue
        m = HUNK_RE.match(ln)
        if m and cur is not None:
            o_start, o_cnt = int(m.group(1)), int(m.group(2) or 1)
            n_cnt = int(m.group(4) or 1)
            if cur['hunks'] == 0 and o_start == 0 and o_cnt == 0:
                cur['is_new'] = True
            cur['hunks'] += 1
            cur['ranges'].append((o_start, o_cnt))
            if m.group(5).strip():
                cur['ctx_symbols'].append(m.group(5).strip())
            ro, rn, i = o_cnt, n_cnt, i + 1
            while i < len(lines) and (ro > 0 or rn > 0):
                l2 = lines[i]
                if l2.startswith('+'):
                    cur['add'] += 1; rn -= 1
                    d = re.match(r'\+\s*(?:async\s+)?(?:def|class)\s+(\w+)', l2)
                    if d: cur['added_defs'].append(d.group(1))
                elif l2.startswith('-'):
                    cur['dele'] += 1; ro -= 1
                elif l2.startswith('\\'):
                    pass
                else:
                    ro -= 1; rn -= 1
                i += 1
            continue
        i += 1
    return files

def kind(path):
    p = path.lower()
    if re.search(r'(^|/)tests?/|test_[^/]*\.py$|_test\.py$|conftest\.py$', p): return 'test'
    if p.startswith(('docs/', 'docs_src/')) or p.endswith(('.md', '.rst', '.txt')): return 'docs'
    if p.startswith(('scripts/', '.github/')): return 'scripts/ci'
    if p.endswith('.py'): return 'source'
    return 'other'

gold = tasks.patch.map(parse_patch)
tests_parsed = tasks.test_patch.map(parse_patch)

def agg(files):
    return pd.Series(dict(
        n_files=len(files), n_hunks=sum(f['hunks'] for f in files), add=sum(f['add'] for f in files),
        dele=sum(f['dele'] for f in files), n_new_files=sum(f['is_new'] for f in files),
        n_deleted_files=sum(f['is_deleted'] for f in files),
        n_source_files=sum(kind(f['path']) == 'source' for f in files),
        n_docs_files=sum(kind(f['path']) == 'docs' for f in files)))

## 6. Localization difficulty: does the issue name the code to change?

# CELL 16 markdown
The agent cannot see the gold patch, so its hardest step is often *finding where to edit*. For each task we check whether the problem statement names the gold file, its module, or any function or class the gold patch edits or adds. Caveat: symbols come from hunk-header context (the enclosing or preceding `def`/`class`) plus newly added definitions, so `symbol named` is an upper bound.

# CELL 17 code
def symbols_from(files):
    syms = set()
    for f in files:
        for c in f['ctx_symbols']:
            m = re.search(r'(?:def|class)\s+(\w+)', c)
            if m: syms.add(m.group(1))
        syms.update(f['added_defs'])
    return {s for s in syms if len(s) > 3 and s not in {'__init__', 'test', 'main', 'self'}}

loc = []
for t, fl in zip(tasks.itertuples(), gold):
    ps_l = t.problem_statement
    src = [f for f in fl if kind(f['path']) in ('source', 'scripts/ci')] or fl
    paths = [f['path'] for f in src]
    base = [os.path.basename(p) for p in paths]
    mods = [p[:-3].replace('/', '.') for p in paths if p.endswith('.py')]
    syms = symbols_from(src)
    loc.append(dict(
        loc_path_named=any(p in ps_l for p in paths),
        loc_file_named=any(re.search(r'\b' + re.escape(b) + r'\b', ps_l) for b in base),
        loc_module_named=any(m in ps_l for m in mods),
        loc_symbol_named=any(re.search(r'\b' + re.escape(s) + r'\b', ps_l) for s in syms),
        n_gold_symbols=len(syms)))
L = pd.DataFrame(loc, index=tasks.index)
tasks = pd.concat([tasks.drop(columns=[c for c in L if c in tasks], errors='ignore'), L], axis=1)
tasks['loc_any'] = tasks[['loc_path_named', 'loc_file_named', 'loc_module_named', 'loc_symbol_named']].any(axis=1)

tab = tasks.groupby('repo')[['loc_path_named', 'loc_file_named', 'loc_module_named', 'loc_symbol_named', 'loc_any']].mean().round(2).loc[REPO_ORDER]

def resolvable(query, ids, ids_lower):
    if query in ids: return 'exact'
    if any(i.endswith('.' + query) or i.endswith('/' + query) for i in ids): return 'suffix'
    if query.lower() in ids_lower: return 'icase'
    if any(query in i for i in ids): return 'substring'
    return None

def module_candidates(path):

    for root in ('src.', 'docs_src.', 'lib.'):
        if m.startswith(root): c.add(m[len(root):])
    return c

def scan_snapshot(args):
    iid, path, targets = args
    t0 = time.time(); res = dict(instance_id=iid, n_files=0, bytes_unc=0, git_bytes=0, n_py=0, py_loc=0,
                                 n_test_py=0, target_lines={}, target_chars={}, err=None)
    tset = set(targets)
    try:
        with tarfile.open(path, mode='r|gz') as tf:
            for m in tf:
                if not m.isfile(): continue
                name = m.name[2:] if m.name.startswith('./') else m.name
                res['n_files'] += 1; res['bytes_unc'] += m.size
                if name.startswith('.git/') or '/.git/' in name:
                    res['git_bytes'] += m.size; continue
                want = name in tset or any(name.endswith('/' + p) for p in tset)
                if name.endswith('.py') or want:
                    data = tf.extractfile(m).read()
                    nl = data.count(b'\n')
                    if name.endswith('.py'):
                        res['n_py'] += 1; res['py_loc'] += nl
                        if re.search(r'(^|/)tests?/', name): res['n_test_py'] += 1
                    if want:
                        key = next((p for p in tset if name == p or name.endswith('/' + p)), name)
                        res['target_lines'][key] = nl
                        res['target_chars'][key] = len(data.decode('utf-8', 'replace'))
    except Exception as e:
        res['err'] = repr(e)
    res['secs'] = round(time.time() - t0, 1)
    return res

def window_calls(ranges, win=150):
    # read_file calls needed to see every hunk if the agent jumps straight to it (greedy 150-line windows)
    calls, end = 0, -1
    for s, c in sorted(ranges):
        e = s + max(c, 1) - 1
        if s > end: calls += 1; end = s + win - 1
        while e > end: calls += 1; end += win
    return calls
