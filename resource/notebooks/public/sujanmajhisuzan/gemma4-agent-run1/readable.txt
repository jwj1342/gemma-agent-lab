import litellm, torch, time
import urllib.request as ur
from adk_submission import VllmConfig, VllmServer, discover_adapters
from swegemma.config import ALLOWED_ADAPTER_EXTENSIONS
from swegemma.models.discovery import validate_single_declared_model
litellm.drop_params=True
INFERENCE_API_KEY='EMPTY'
TARGET_MODEL_NAME='gemma-4-31b-it-qat-w4a16-ct'
MODEL_PATH=Path('/kaggle/input/models/google/gemma-4/other/gemma-4-31b-it-qat-w4a16-ct/2')
print('model path exists:',MODEL_PATH.exists())
declared_model=validate_single_declared_model(AGENT_DIR)
adapters=discover_adapters(str(AGENT_DIR),adapter_extensions=ALLOWED_ADAPTER_EXTENSIONS)
gpu_count=torch.cuda.device_count() if torch.cuda.is_available() else 1
cap=torch.cuda.get_device_capability(0) if torch.cuda.is_available() else (0,0)
print('gpu count:',gpu_count,'capability:',cap)
# T4 is compute 7.5 -> no bf16. Select dtype from capability, not is_bf16_supported().
dtype='bfloat16' if cap[0]>=8 else 'float16'
tp_size=4 if gpu_count>=4 else (2 if gpu_count>=2 else 1)
# tp must stay 2: the W4A16 weights are ~18GB and one T4 has only 16GB.
# vLLM's custom all-reduce kernel fails on T4 with
#   Cuda error custom_all_reduce.cuh:455 'invalid argument' -> Engine core init failed.
# Documented fix (vllm-project/vllm#45604): fall back to NCCL native collectives.
EXTRA_ARGS=['--disable-custom-all-reduce']
if cap[0]<8:
    EXTRA_ARGS+=['--enforce-eager','--max-num-seqs','4']
print('dtype',dtype,'tp',tp_size,'extra_args',EXTRA_ARGS)
# gpu_memory_utilization: 0.90 OOMed on 2xT4 (24GB total, ~18GB weights). 0.95 fits.
vllm_cfg=VllmConfig(
    model=str(MODEL_PATH),
    port=8000,
    host='127.0.0.1',
    tool_call_parser='gemma4',
    reasoning_parser='gemma4',
    max_model_len=32768,
    dtype=dtype,
    gpu_memory_utilization=0.95,
    enable_auto_tool_choice=True,
    enable_lora=True,
    max_loras=8,
    max_lora_rank=128,
    tensor_parallel_size=tp_size,
    startup_timeout=60*20,
    extra_args=EXTRA_ARGS,
)
# Single start, never stopped: the retry ladder's server.stop() tore down the
# port/GPU state and left the healthy server unreachable from the agent.
server_instance=VllmServer(vllm_cfg,adapter_manifest=adapters)
server_instance.start()
print(f'vLLM server started on {server_instance.base_url} (tp={tp_size})')
models=server_instance.create_model_registry(
    aliases=[declared_model,TARGET_MODEL_NAME],
    model_prefix='openai/',
    api_key=INFERENCE_API_KEY,
)
try:
    r=ur.urlopen('http://127.0.0.1:8000/v1/models',timeout=15)
    print('HEALTH 8000',r.status)
except Exception as e:
    print('HEALTH FAILED',type(e).__name__,str(e)[:160]); raise


# CELL 4 code
import asyncio, concurrent.futures, pandas as pd, urllib.request, yaml
from pathlib import Path
from google.adk.agents.context_cache_config import ContextCacheConfig
from google.adk.apps._configs import EventsCompactionConfig
from swegemma.config import EvalConfig, build_submission_limits
from swegemma.evaluate import Evaluator

def run_sync(fn,*a,**k):
    try: loop=asyncio.get_running_loop()
    except RuntimeError: loop=None
    if loop and loop.is_running():
        with concurrent.futures.ThreadPoolExecutor(max_workers=1) as p:
            return p.submit(lambda: asyncio.run(fn(*a,**k))).result()
    return asyncio.run(fn(*a,**k))

# v2 ships NO eval_config.yaml -> defaults (100 calls / 60 min / 300s / 500 turns)
timeout_seconds,max_tool_calls,max_time_minutes,max_turns=300,100,60.0,None

limits,gen=build_submission_limits()
eval_config=EvalConfig(tasks_path=TASKS_PATH,snapshots_dir=DATA_DIR/'snapshots',results_dir=WORKING_DIR/'results',
    submission_dir=AGENT_DIR,models=models,sandbox='subprocess',timeout_seconds=timeout_seconds,
    max_time_minutes=max_time_minutes,max_tool_calls=max_tool_calls,max_turns=max_turns,
    limits=limits,generation_constraints=gen,adapter_manifest=adapters,
    context_cache_config=ContextCacheConfig(min_tokens=2048,ttl_seconds=1800,cache_intervals=10),
    events_compaction_config=EventsCompactionConfig(compaction_interval=15,overlap_size=2,
       token_threshold=14336,event_retention_size=5),
    graph_dir=str(DATA_DIR/'graphs'),embeddings_dir=str(DATA_DIR/'embeddings'),
    wheels_dir=DATA_DIR/'wheels',verbose=False)

ev=Evaluator(eval_config)
rows=[]
for idx,task in enumerate(SAMPLE_TASKS,1):
    # liveness re-check: if vLLM died mid-run, stop now instead of scoring 8x Connection error
    alive=False
    try:
        alive=urllib.request.urlopen('http://127.0.0.1:8000/v1/models',timeout=10).status==200
    except Exception as _e:
        print('  !! vLLM NOT REACHABLE before task',idx,type(_e).__name__)
    if not alive:
        print('  server log tail:'); 
        try: print(Path(server_instance.log_path).read_text()[-2000:])
        except Exception: pass
        break
    print(f'\n===== [{idx}/{len(SAMPLE_TASKS)}] {task.instance_id}')
    try:
        r=run_sync(ev.evaluate_task,task=task,task_index=idx,total_tasks=len(SAMPLE_TASKS))
        print(f'  resolved={r.resolved} exit={r.test_exit_code} calls={r.tool_calls} {r.duration_seconds:.1f}s patch={len(r.agent_patch or "")}B err={(r.error or "")[:200]}')
        rows.append({'id':task.instance_id,'prediction':r.agent_patch or '',"resolved":bool(r.resolved),'exit':r.test_exit_code,'calls':r.tool_calls,'secs':float(r.duration_seconds),'err':(r.error or "")[:400]})
    except Exception as e:
        import traceback; traceback.print_exc()
        rows.append({'id':task.instance_id,'prediction':'','resolved':False,'exit':None,'calls':0,'secs':0.0,'err':repr(e)[:400]})

df=pd.DataFrame(rows)
rn=int(df['resolved'].sum())
print(f'\n===== RESULT: {rn}/{len(df)} resolved  rate={rn/len(df):.3f}')
display(df[['id','resolved','exit','calls','secs']])
