HUNK_RE = re.compile(r"^@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@(.*)$")
SYMBOL_RE = re.compile(r"(?:async\s+)?(?:def|class)\s+([A-Za-z_]\w*)")

def parse_patch(text):
    files, cur = [], None
    rem_old = rem_new = malformed = 0

    def new_file(path):
        f = {"path": path, "added": 0, "deleted": 0, "hunks": 0, "symbols": set(), "status": "modified"}
        files.append(f)
        return f

    for line in (text or "").splitlines():
        if rem_old > 0 or rem_new > 0:
            tag = line[:1]
            if tag == "+":
                cur["added"] += 1; rem_new -= 1
                m = SYMBOL_RE.match(line[1:].strip()); m and cur["symbols"].add(m.group(1))
                continue
            if tag == "-":
                cur["deleted"] += 1; rem_old -= 1
                m = SYMBOL_RE.match(line[1:].strip()); m and cur["symbols"].add(m.group(1))
                continue
            if tag == " " or line == "":
                rem_old -= 1; rem_new -= 1
                continue
            malformed += 1
            rem_old = rem_new = 0
            
        if line.startswith("+++ "):
            path = line[4:].split("\t")[0].strip()
            path = path[2:] if path.startswith("b/") else path
            cur = new_file(path)
            continue
            
        m = HUNK_RE.match(line)
        if m and cur is not None:
            rem_old = int(m.group(2)) if m.group(2) is not None else 1
            rem_new = int(m.group(4)) if m.group(4) is not None else 1
            cur["hunks"] += 1

    for f in files:
        f["symbols"] = sorted(f["symbols"])
    return {"files": files, "malformed": malformed, "unterminated": rem_old > 0 or rem_new > 0}

if DATA_DIR and (DATA_DIR / "tasks.jsonl").exists():
    with open(DATA_DIR / "tasks.jsonl", "r", encoding="utf-8") as f:

    print(f"Loaded {len(df)} tasks. Median churn: {int(df['churn'].median())} lines.")

def visualize_code_graph(instance_id="fastapi_11194", max_nodes=25):
    """Renders the AST call and dependency graph for a specific task."""
    graph_path = DATA_DIR / "graphs" / f"{instance_id}.json" if DATA_DIR else None
    G = nx.DiGraph()
    
    if graph_path and graph_path.exists():
        with open(graph_path, "r", encoding="utf-8") as f:
            data = json.load(f)
        nodes = data.get("nodes", [])[:max_nodes]
        node_ids = {n["id"] for n in nodes}
        for n in nodes:
            G.add_node(n["id"].split(".")[-1])
        for edge in data.get("edges", []):
            if edge["source"] in node_ids and edge["target"] in node_ids:
                G.add_edge(edge["source"].split(".")[-1], edge["target"].split(".")[-1])
                
        plt.figure(figsize=(10, 6))
        nx.draw_networkx(G, nx.spring_layout(G, seed=42), node_size=1000, node_color="lightblue", font_size=8)
        plt.title(f"AST Symbol Call Graph: {instance_id}")
        plt.axis("off")
        plt.show()

if DATA_DIR:
    visualize_code_graph()

def write_archive(files: dict, output_path: Path):
    """Validates files against ADK schema limits and creates submission.zip."""
    allowed = {'.yaml', '.yml', '.md', '.txt', '.py', '.json', '.safetensors'}
    
    with zipfile.ZipFile(output_path, 'w', compression=zipfile.ZIP_DEFLATED) as archive:
        for name, content in sorted(files.items()):
            path = Path(name)
            assert path.suffix in allowed, f"Invalid extension: {path.suffix}"
            assert not path.is_absolute() and '..' not in path.parts, f"Path traversal blocked: {name}"
            archive.writestr(name, content.encode('utf-8'))
            
    print(f"Successfully packaged {len(files)} files into {output_path.name} ({output_path.stat().st_size} bytes).")

write_archive(BUNDLE_FILES, WORKING / "submission.zip")
