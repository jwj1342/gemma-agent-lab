# CELL 8 code
"""Check a submission directory or zip against the HARNESS_README rules.

This approximates the official validator (adk-submission), which is not
installed locally. Usage: python tools/check_submission.py submission.zip
"""
import re
import sys
import tempfile
import zipfile
from pathlib import Path

import yaml

TOOLS = {"run_command", "submit_patch", "get_status", "read_file", "edit_file", "write_file",
         "get_code_neighbors", "search_similar_code", "get_code_subgraph"}
EXTS = {".yaml", ".yml", ".md", ".txt", ".py", ".json", ".safetensors"}
GEN_KEYS = {"temperature", "top_p", "top_k", "max_output_tokens", "presence_penalty", "frequency_penalty",
            "stop_sequences", "response_mime_type", "seed", "thinking_config"}
STATE_VARS = {"problem_description", "hints"}
ROOT_NAMES = ("agent.yaml", "agent.yml", "root_agent.yaml", "root_agent.yml")


class Report:
    def __init__(self):
        self.errors, self.notes = [], []


def _loader(report, root, base, depth):
    class Loader(yaml.SafeLoader):
        pass

    def include(loader, node):
        rel = loader.construct_scalar(node)
        if rel.startswith("/") or "\0" in rel or re.match(r"^[A-Za-z]:", rel):
            report.errors.append(f"absolute include: {rel}")
            return None
        target = (base / rel).resolve()
        if root not in target.parents:
            report.errors.append(f"include escapes root: {rel}")
            return None
        if not target.is_file():
            report.errors.append(f"include missing: {rel} (from {base.name}/)")
            return None
        if target.suffix in (".md", ".txt"):
            return target.read_text(encoding="utf-8")
        return _load(report, root, target, depth + 1)

    Loader.add_constructor("!include", include)
    return Loader


def _load(report, root, path, depth=0):
    if depth > 10:
        report.errors.append("include depth > 10")
        return None
    return yaml.load(path.read_text(encoding="utf-8"), Loader=_loader(report, root, path.parent, depth))


def _check_agent(report, root, cfg, where, models):
    if not isinstance(cfg, dict):
        report.errors.append(f"{where}: not a mapping")
        return
    if cfg.get("model"):
        models.add(cfg["model"].split("/")[-1])
    if cfg.get("adapter") and not (root / "adapters" / cfg["adapter"] / "adapter_model.safetensors").is_file():
        report.errors.append(f"{where}: adapter {cfg['adapter']} missing")
    instruction = cfg.get("instruction") or ""
    for var in re.findall(r"\{+([^{}]*)\}+", instruction):
        name = var.strip().rstrip("?")
        if name.isidentifier() and name not in STATE_VARS and not var.endswith("?"):
            report.errors.append(f"{where}: instruction placeholder {{{var}}} is not in session state")
    gcc = cfg.get("generate_content_config") or {}
    for key in gcc:
        if key not in GEN_KEYS:
            report.errors.append(f"{where}: generate_content_config.{key} not allowed")
    if not 1 <= gcc.get("max_output_tokens", 16384) <= 32768:
        report.errors.append(f"{where}: max_output_tokens out of range")
    budget = (gcc.get("thinking_config") or {}).get("thinking_budget", 4096)
    if not 1 <= budget <= 32768:
        report.errors.append(f"{where}: thinking_budget out of range")
    for tool in cfg.get("tools") or []:
        if isinstance(tool, str):
            if tool not in TOOLS:
                report.errors.append(f"{where}: unknown tool {tool}")
        elif isinstance(tool, dict) and "agent_tool" in tool:
            config_path = tool["agent_tool"].get("config_path")
            path = root / config_path
            if not path.is_file():
                report.errors.append(f"{where}: agent_tool config_path missing: {config_path}")
            else:
                _check_agent(report, root, _load(report, root, path), config_path, models)
        else:
            report.errors.append(f"{where}: unrecognised tool entry {tool!r}")
    for sub in cfg.get("sub_agents") or []:
        if not (isinstance(sub, dict) and set(sub) == {"config_path"}):
            keys = list(sub) if isinstance(sub, dict) else sub
            report.errors.append(f"{where}: sub_agents entry must be {{config_path: ...}}, got {keys}")
            continue
        path = root / sub["config_path"]
        if not path.is_file():
            report.errors.append(f"{where}: sub_agent config_path missing: {sub['config_path']}")
        else:
            _check_agent(report, root, _load(report, root, path), sub["config_path"], models)
    for skill in cfg.get("skills") or []:
        skill_md = root / skill / "SKILL.md"
        if not skill_md.is_file():
            report.errors.append(f"{where}: skill dir {skill} has no SKILL.md")
            continue
        front = yaml.safe_load(skill_md.read_text(encoding="utf-8").split("---")[1])
        if not re.fullmatch(r"[a-z0-9]+(-[a-z0-9]+)*", front.get("name", "")):
            report.notes.append(f"{where}: skill name {front.get('name')!r} is not kebab-case (needs ADK feature flag)")


def check_dir(root, report=None):
    report = report or Report()
    root = Path(root).resolve()
    roots = [n for n in ROOT_NAMES if (root / n).is_file()]
    if len(roots) != 1:
        report.errors.append(f"expected exactly one root config, found {roots}")
        return report
    total = 0
    for f in root.rglob("*"):
        if f.is_symlink():
            report.errors.append(f"symlink: {f.relative_to(root)}")
        if f.is_file():
            total += f.stat().st_size
            if f.suffix not in EXTS:
                report.errors.append(f"disallowed extension: {f.relative_to(root)}")
    if total >= 3 * 1024**3:
        report.errors.append(f"total size {total} >= 3 GiB")
    models = set()
    _check_agent(report, root, _load(report, root, root / roots[0]), roots[0], models)
    if len(models) != 1:
        report.errors.append(f"expected one base model, found {models}")
    eval_config = root / "eval_config.yaml"
    report.notes.append("eval_config: " + (str(yaml.safe_load(eval_config.read_text())) if eval_config.is_file()
                                           else "not included (harness defaults)"))
    report.notes.append(f"files={sum(1 for f in root.rglob('*') if f.is_file())} size={total} bytes models={models}")
    return report


def check(target):
    """Check a directory or .zip; print the report and return True if it passes."""
    target = Path(target)
    report = Report()
    if target.suffix == ".zip":
        with zipfile.ZipFile(target) as archive:
            names = archive.namelist()
            bad = [n for n in names if "\\" in n or n.startswith("/")]
            if bad:
                report.errors.append(f"bad zip entry names: {bad}")
            report.notes.append(f"zip entries: {sorted(names)}")
            extracted = Path(tempfile.mkdtemp())
            archive.extractall(extracted)
        check_dir(extracted, report)
    else:
        check_dir(target, report)
    for note in report.notes:
        print("NOTE ", note)
    for error in report.errors:
        print("ERROR", error)
    print("RESULT:", "FAIL" if report.errors else "PASS")
    return not report.errors
