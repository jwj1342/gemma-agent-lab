# cgr - Code Graph Repair for the Gemma 4 Developer Agent corpus

Repairs and re-indexes the competition's `graphs/` + `embeddings/` release, then
measures what the shipped defects cost an agent.

* **Defect 1 - hard-link loss.** `graphs/` and `embeddings/` ship 256 files each
  (129 task-named hard-linked to 127 commit-named). ZIP cannot store hard links,
  so exactly one name per commit group survives the download and the rest arrive
  0 bytes. The harness enables the code-intelligence tools only when those files
  exceed 100 bytes, so `get_code_neighbors`, `get_code_subgraph` and
  `search_similar_code` silently switch off.
* **Defect 2 - no `async` nodes.** Graph generation enumerates `def` and `class`
  but not `async def`, so asynchronous functions are absent from the node set
  and are unreachable by all three graph tools.

Everything runs on CPU. No LLM inference, no GPU, no network after download.

**How to run.** On Kaggle, attach the `gemma-4-developer-agent` competition data as
an input and *Run All*; the read-only input is staged into `/kaggle/working`
automatically. On Colab, upload your `kaggle.json` and make sure ~50 GB is free.
`results.json` is written to the working directory at the end.

The tool is embedded in the cell below, so this notebook is standalone.

"""
cgr - Code Graph Repair
=======================

Audit, repair, and re-index the Gemma 4 Developer Agent Competition
code-graph / node-embedding corpus.

Motivation
----------
The competition ships ``graphs/`` and ``embeddings/`` as 256 files each: 129
task-named files hard-linked to 127 commit-named files for the same commit.
ZIP archives cannot store hard links, so on download exactly one name per
commit group survives with content and the other arrives 0 bytes. The harness
only enables the code-intelligence tools when those files exceed 100 bytes, so
the three graph tools silently vanish for a large share of tasks.

Separately, graph generation enumerates only ``def``/``class`` nodes, so
``async def`` bodies are absent from the node set and therefore unreachable by
``get_code_neighbors``, ``get_code_subgraph`` and ``search_similar_code``.

This module provides five tools:

  1. ``census()``     - quantify the corruption.
  2. ``repair()``     - rebuild every commit group from its surviving copy.
  3. ``audit()``      - measure graph node coverage against the snapshots.
  4. ``reindex()``    - add the missing (mostly async) nodes; no vectors.
  5. ``retrieval_eval()`` - measure what the corruption costs an agent, using
                        only the released 256-d vectors. No LLM required.

Everything runs on CPU. The only third-party dependency is numpy, needed to
read the released ``.npz`` embeddings.

Public entry point
------------------
    python cgr.py report --root /kaggle/input/... --out results/   # MUTATES IN PLACE

CLI
---
    python cgr.py census   --root ROOT
    python cgr.py repair   --root ROOT [--dry-run]
    python cgr.py audit    --root ROOT --out OUT
    python cgr.py reindex  --root ROOT --out OUT [--no-embed]
    python cgr.py bench    --root ROOT --out OUT [--k 1 5 10]
    python cgr.py report   --root ROOT --out OUT     # all of the above
"""

# Harness constants

#: HARNESS_README gates the code-intelligence tools on a file-size check.
GRAPH_MIN_BYTES = 100

#: Released embedding dimensionality (float32, one vector per graph node).
EMBED_DIM = 256

#: Files whose names look like tests are excluded from coverage auditing.
_TEST_RE = re.compile(r"(^|[/_.])(tests?|testing)([/_.]|$)|(^|[/_.])test_[^/]*\.py$|_test\.py$|conftest\.py$", re.I)

_TASK_FILE = "tasks.jsonl"


# --------------------------------------------------------------------------
# Small helpers
# --------------------------------------------------------------------------


def _repo_short(repo: str) -> str:
    """``psf/requests`` -> ``requests`` (the stem used in graph/embedding names)."""
    return repo.rstrip("/").rsplit("/", 1)[-1]


def _commit_key_for_task(task: dict) -> str:
    """``fastapi/fastapi`` @ ``<sha>`` -> ``fastapi_<sha>``."""
    return f"{_repo_short(task['repo'])}_{task['base_commit']}"


def _load_tasks(root: Path) -> List[dict]:
    path = root / _TASK_FILE
    if not path.exists():
        raise FileNotFoundError(f"missing {_TASK_FILE} under {root}")
    out = []
    with path.open(encoding="utf-8") as fh:
        for line in fh:
            line = line.strip()
            if line:
                out.append(json.loads(line))
    return out


def load_tasks(root: str | Path) -> List[dict]:
    """Public entry point: the benchmark instances, as dicts."""
    return _load_tasks(Path(root))


def _is_empty_file(p: Path) -> bool:
    try:
        return p.stat().st_size <= GRAPH_MIN_BYTES
    except OSError:
        return True


def _sha256(p: Path) -> str:
    import hashlib

    h = hashlib.sha256()
    with p.open("rb") as fh:
        for chunk in iter(lambda: fh.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


# --------------------------------------------------------------------------
# Grouping: which files are supposed to be hard links of each other?
# --------------------------------------------------------------------------


@dataclass
class Group:
    """One commit's worth of alias names for a single logical artifact."""

    key: str
    names: List[str] = field(default_factory=list)
    readable: List[str] = field(default_factory=list)
    #: unusable names: 0-byte (or <= GRAPH_MIN_BYTES) files *and* absent task files
    dead: List[str] = field(default_factory=list)
    #: the subset of ``dead`` that is not on disk at all
    missing: List[str] = field(default_factory=list)
    #: the surviving copy landed under a task name belonging to another commit
    misassigned: bool = False
    #: the surviving copy landed under a *sibling* task of the same commit
    sibling_named: bool = False


def build_groups(
    files: Iterable[str], tasks_by_id: Dict[str, dict], ext: str
) -> Dict[str, Group]:
    """Map every filename onto its commit group.

    A task-named file ``<instance_id>.<ext>`` belongs to the group
    ``<repo_short>_<base_commit>``. A commit-named file already *is* a group
    key. Names that belong to neither are placed in a singleton group.
    """
    groups: Dict[str, Group] = {}

    def get(key: str) -> Group:
        return groups.setdefault(key, Group(key=key))

    for fname in files:
        stem = fname[: -len(ext)] if fname.endswith(ext) else Path(fname).stem
        if stem in tasks_by_id:
            g = get(_commit_key_for_task(tasks_by_id[stem]))
            g.names.append(fname)
        elif "_" in stem:
            g = get(stem)
            g.names.append(fname)
        else:
            g = get(stem)
            g.names.append(fname)
    return groups


def _scan_dir(path: Path, ext: str, tasks_by_id: Dict[str, dict]) -> Dict[str, Group]:
    """Group the files in ``path`` by commit.

    Every task's ``<instance_id><ext>`` is an expected member of its group even
    when it is absent: depending on how the release was unpacked, the lost
    hard-link aliases arrive either as 0-byte files (a local unzip) or not at
    all (the mounted Kaggle input).
    """
    if not path.is_dir():
        return {}
    on_disk = [p.name for p in path.iterdir() if p.name.endswith(ext)]
    have = set(on_disk)
    expected = [f"{iid}{ext}" for iid in tasks_by_id if f"{iid}{ext}" not in have]
    groups = build_groups(on_disk + expected, tasks_by_id, ext)
    for key, g in groups.items():
        for fname in g.names:
            fp = path / fname
            if fname not in have:
                g.dead.append(fname)
                g.missing.append(fname)
            elif _is_empty_file(fp):
                g.dead.append(fname)
            else:
                g.readable.append(fname)
        # A survivor is "misassigned" if it sits under a task name whose commit
        # differs from this group, and "sibling-named" if it sits under another
        # task of the *same* commit (content-equivalent, but invisible to any
        # per-task file check).
        member_ids = {iid for iid, t in tasks_by_id.items() if _commit_key_for_task(t) == key}
        for n in g.readable:
            stem = Path(n).stem
            if stem in tasks_by_id and stem not in member_ids:
                g.misassigned = True
            elif stem in tasks_by_id and len(member_ids) > 1:
                g.sibling_named = True
    return groups


# --------------------------------------------------------------------------
# 1. Census - quantify the corruption
# --------------------------------------------------------------------------


def census(root: str | Path) -> dict:
    root = Path(root)
    tasks = _load_tasks(root)
    by_id = {t["instance_id"]: t for t in tasks}

    report: dict = {
        "root": str(root),
        "n_tasks": len(tasks),
        "repos": sorted({t["repo"] for t in tasks}),
        "graph": {},
        "embeddings": {},
        "mislabelled_groups": [],
        "note": (f"a file counts as dead when size <= {GRAPH_MIN_BYTES} bytes; a task file that is "
                 "absent is counted in n_missing_task_files, and both disable the task's tools"),
    }

    for kind, sub, ext in (("graph", "graphs", ".json"), ("embeddings", "embeddings", ".npz")):
        groups = _scan_dir(root / sub, ext, by_id)
        n_missing = sum(len(g.missing) for g in groups.values())
        n_files = sum(len(g.names) for g in groups.values()) - n_missing  # on disk
        n_dead = sum(len(g.dead) for g in groups.values()) - n_missing  # on disk, <= 100 bytes
        shared = [g for g in groups.values() if len(g.names) > 1]
        ambiguous = [g.key for g in shared if len(g.readable) != 1]

        # Per-task impact: a task whose OWN file is 0 bytes has the graph tools
        # disabled, even when its group still holds a readable copy elsewhere.
        dead_own, recoverable, lost = [], [], []
        for g in groups.values():
            for fname in g.names:
                stem = Path(fname).stem
                if stem not in by_id:
                    continue
                if fname in g.dead:  # 0-byte or absent
                    dead_own.append(stem)
                    (recoverable if g.readable else lost).append(stem)
        n = len(by_id)
        report[kind] = {
            "n_files": n_files,
            "n_groups": len(groups),
            "n_shared_groups": len(shared),
            "n_dead_files": n_dead,
            "pct_files_dead": round(100.0 * n_dead / n_files, 2) if n_files else 0.0,
            "n_missing_task_files": n_missing,
            "pct_groups_with_no_survivor": round(
                100.0 * len([g for g in shared if not g.readable]) / len(shared), 2
            )
            if shared
            else 0.0,
            "n_groups_not_exactly_one_survivor": len(ambiguous),
            "n_tasks": n,
            "n_tasks_own_file_dead": len(dead_own),
            "pct_tasks_graph_tools_disabled": round(100.0 * len(dead_own) / n, 2) if n else 0.0,
            "n_tasks_recoverable_via_group": len(recoverable),
            "n_tasks_unrecoverable": len(lost),
            "n_sibling_named_survivors": sum(1 for g in groups.values() if g.sibling_named),
        }
        if lost:
            report.setdefault("unrecoverable_tasks", {})[kind] = lost
        for g in groups.values():
            if g.misassigned:
                report["mislabelled_groups"].append(
                    {"artifact": kind, "group": g.key, "survivor": g.readable, "names": g.names}
                )
    return report


# --------------------------------------------------------------------------
# 2. Repair - rebuild every commit group from its surviving copy
# --------------------------------------------------------------------------


def repair(root: str | Path, dry_run: bool = False) -> dict:
    """Copy each commit's one readable artifact over its 0-byte or absent aliases.

    Idempotent: already-consistent groups are left untouched. Groups with no
    survivor are reported, never fabricated.
    """
    root = Path(root)
    tasks = _load_tasks(root)
    by_id = {t["instance_id"]: t for t in tasks}

    out: dict = {"dry_run": dry_run, "repaired": [], "unrecoverable": [], "already_ok": 0}

    for kind, sub, ext in (("graph", "graphs", ".json"), ("embeddings", "embeddings", ".npz")):
        d = root / sub
        groups = _scan_dir(d, ext, by_id)
        for g in groups.values():
            if not g.readable:
                out["unrecoverable"].append({"artifact": kind, "group": g.key, "names": g.names})
                continue
            if not g.dead:
                out["already_ok"] += 1
                continue
            src = d / g.readable[0]
            digest = _sha256(src)
            for name in g.dead:
                rec = {
                    "artifact": kind,
                    "group": g.key,
                    "from": g.readable[0],
                    "to": name,
                    "sibling_named_survivor": g.sibling_named,
                    "was_missing": name in g.missing,
                }
                out["repaired"].append(rec)
                if not dry_run:
                    shutil.copyfile(src, d / name)
                    if _sha256(d / name) != digest:
                        raise IOError(f"copy verification failed for {d / name}")
    out["n_repaired"] = len(out["repaired"])
    out["n_unrecoverable"] = len(out["unrecoverable"])
    return out

def load_graph(path: str | Path) -> dict:
    with Path(path).open(encoding="utf-8") as fh:
        return json.load(fh)


def graph_nodes(g: dict) -> List[str]:
    return [n["id"] for n in g.get("nodes", [])]


def load_embeddings(path: str | Path) -> Dict[str, "object"]:
    import numpy as np

    with np.load(Path(path), allow_pickle=True) as z:
        return {k: z[k] for k in z.files}


def _as_matrix(emb: Dict[str, Any]):
    """Stack an embedding dict into (node_ids, L2-normalised float32 matrix)."""
    import numpy as np

    ids = sorted(emb)
    if not ids:
        return [], np.zeros((0, EMBED_DIM), dtype=np.float32)
    mat = np.stack([np.asarray(emb[i], dtype=np.float32).reshape(-1) for i in ids])
    norms = np.linalg.norm(mat, axis=1, keepdims=True)
    norms[norms == 0] = 1.0
    return ids, mat / norms

def _iter_snapshot_files(tgz: Path, tmp: Path) -> Path:
    """Extract the Python sources of one snapshot; return the repo root directory.

    The snapshots carry full git history, but cgr only ever reads ``*.py``
    files outside ``.git``, so only those are written. The archive is streamed
    in a single pass (random access into a ``.tgz`` re-decompresses it).
    """
    tops: Set[str] = set()
    nested = False
    with tarfile.open(tgz, "r|gz") as tf:
        for m in tf:
            if m.name.startswith("/"):
                continue  # no absolute paths
            parts = Path(m.name).parts
            if not parts or ".." in parts:
                continue  # no path traversal out of the destination
            if not parts[0].startswith("."):
                tops.add(parts[0])
                nested = nested or len(parts) > 1
            if not m.isfile() or not parts[-1].endswith(".py") or ".git" in parts:
                continue
            src = tf.extractfile(m)
            if src is None:
                continue
            dest = tmp.joinpath(*parts)
            dest.parent.mkdir(parents=True, exist_ok=True)
            with src, dest.open("wb") as out:
                shutil.copyfileobj(src, out)
    # snapshots are a bare git worktree, but tolerate a single wrapper dir
    if len(tops) == 1 and nested and (tmp / next(iter(tops))).is_dir():
        return tmp / next(iter(tops))
    return tmp


def _progress(label: str, i: int, n: int) -> None:
    """One stderr line every 10 tasks, so a long notebook cell shows it is alive."""
    if i == n or i % 10 == 0:
        print(f"  {label}: {i}/{n} tasks", file=sys.stderr, flush=True)


def _module_name(rel: str) -> str:
    """``src/fastapi/routing.py`` -> ``fastapi.routing``; ``pkg/__init__.py`` -> ``pkg``."""
    p = Path(rel)
    parts = list(p.parts)
    if parts and parts[0] in ("src", "lib"):
        parts = parts[1:]
    if not parts:
        return ""
    if parts[-1] == "__init__.py":
        parts = parts[:-1]
    else:
        parts[-1] = parts[-1][: -len(".py")]
    return ".".join(x for x in parts if x)


def _is_test_path(rel: str) -> bool:
    return bool(_TEST_RE.search(rel))


@dataclass
class Symbol:
    node_id: str
    kind: str  # "sync_function" | "async_function" | "class"
    is_async: bool
    file: str
    lineno: int
    end_lineno: int = 0


def enumerate_symbols(root_dir: Path, files: Optional[Sequence[str]] = None) -> List[Symbol]:
    """Every module-level class/function (sync and async) in the Python sources.

    This is the reference set that the graph should have covered.
    """
    py_files = []
    for p in sorted(root_dir.rglob("*.py")):
        rel = str(p.relative_to(root_dir))
        if ".git" in rel.split(os.sep) or _is_test_path(rel):
            continue
        py_files.append((p, rel))

    if files is not None:
        wanted = {f.lstrip("./") for f in files}
        py_files = [(p, r) for p, r in py_files if r in wanted or r.lstrip("./") in wanted]

    out: List[Symbol] = []
    for path, rel in py_files:
        try:
            src = path.read_text(encoding="utf-8", errors="replace")
            tree = ast.parse(src)
        except (SyntaxError, ValueError, UnicodeDecodeError):
            continue
        mod = _module_name(rel)
        if not mod:
            continue
        for node in tree.body:
            if isinstance(node, ast.ClassDef):
                out.append(Symbol(f"{mod}.{node.name}", "class", False, rel, node.lineno, _span(node)[1]))
                for sub in node.body:
                    if isinstance(sub, (ast.FunctionDef, ast.AsyncFunctionDef)):
                        out.append(_func_symbol(f"{mod}.{node.name}.{sub.name}", sub, rel))
            elif isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                out.append(_func_symbol(f"{mod}.{node.name}", node, rel))
    return out


def _span(node: ast.AST) -> Tuple[int, int]:
    """First and last source line of a definition, decorators included."""
    lo = min([node.lineno] + [d.lineno for d in getattr(node, "decorator_list", [])])
    return lo, getattr(node, "end_lineno", None) or node.lineno


def _func_symbol(node_id: str, node: ast.AST, rel: str) -> Symbol:
    is_async = isinstance(node, ast.AsyncFunctionDef)
    kind = "async_function" if is_async else "sync_function"
    return Symbol(node_id, kind, is_async, rel, node.lineno, _span(node)[1])


def _skip_reason(tgz: Path, gpath: Path) -> Optional[str]:
    """Why a task cannot be measured, or None if it can."""
    if not tgz.exists():
        return "no_snapshot"
    if _is_empty_file(gpath):
        return "dead_or_missing_graph"  # run repair() first
    return None


def _extract(tgz: Path, td: str, iid: str) -> Tuple[Optional[Path], Optional[str]]:
    """Extract one snapshot into a fresh per-task directory under ``td``."""
    sub = Path(td) / iid
    shutil.rmtree(sub, ignore_errors=True)
    sub.mkdir(parents=True)
    try:
        return _iter_snapshot_files(tgz, sub), None
    except (tarfile.TarError, OSError) as e:
        return None, f"bad_snapshot: {type(e).__name__}: {e}"[:200]


def _source_segment(src: str, node: ast.AST) -> str:
    lines = src.splitlines()
    start = getattr(node, "lineno", 1) - 1
    end = getattr(node, "end_lineno", start + 1)
    return "\n".join(lines[start:end])


# --------------------------------------------------------------------------
# 3. Audit - node coverage of the reference symbol set
# --------------------------------------------------------------------------


def audit(root: str | Path, limit: Optional[int] = None) -> dict:
    """Per-task node coverage, broken down by symbol kind and by repository."""
    root = Path(root)
    tasks = _load_tasks(root)
    if limit:
        tasks = tasks[:limit]

    per_task: List[dict] = []
    by_repo: Dict[str, Dict[str, List[int]]] = defaultdict(lambda: defaultdict(list))
    totals: Dict[str, List[int]] = defaultdict(list)
    skipped: Dict[str, str] = {}

    with tempfile.TemporaryDirectory() as td:
        for i, t in enumerate(tasks, 1):
            _progress("audit", i, len(tasks))
            iid = t["instance_id"]
            tgz = root / "snapshots" / f"{iid}.tgz"
            gpath = root / "graphs" / f"{iid}.json"
            why = _skip_reason(tgz, gpath)
            if why:
                skipped[iid] = why
                continue
            repo_dir, why = _extract(tgz, td, iid)
            if repo_dir is None:
                skipped[iid] = why
                continue

            g = load_graph(gpath)
            present: Set[str] = set(graph_nodes(g))
            syms = enumerate_symbols(repo_dir)
            shutil.rmtree(Path(td) / iid, ignore_errors=True)  # 129 full repos will not fit at once

            counts: Dict[str, List[int]] = defaultdict(list)
            missing_examples: List[str] = []
            for s in syms:
                hit = s.node_id in present
                counts[s.kind].append(1 if hit else 0)
                if not hit and s.is_async and len(missing_examples) < 5:
                    missing_examples.append(s.node_id)

            rec = {"instance_id": iid, "repo": t["repo"], "n_graph_nodes": len(present)}
            for kind in ("sync_function", "async_function", "class"):
                vals = counts.get(kind, [])
                found, total = sum(vals), len(vals)
                rec[f"{kind}_found"] = found
                rec[f"{kind}_total"] = total
                by_repo[t["repo"]][kind].extend(vals)
                totals[kind].extend(vals)
            rec["missing_async_examples"] = missing_examples
            per_task.append(rec)

    def _agg(vals: List[int]) -> dict:
        return {"found": sum(vals), "total": len(vals),
                "pct": round(100.0 * sum(vals) / len(vals), 2) if vals else 0.0}

    return {
        "n_tasks_audited": len(per_task),
        "n_tasks_skipped": len(skipped),
        "skip_reasons": _count_reasons(skipped),
        "skipped": dict(list(skipped.items())[:20]),
        "per_task": per_task,
        "overall": {k: _agg(v) for k, v in totals.items()},
        "by_repo": {r: {k: _agg(v) for k, v in kinds.items()} for r, kinds in by_repo.items()},
    }


def _count_reasons(skipped: Dict[str, str]) -> Dict[str, int]:
    out: Dict[str, int] = defaultdict(int)
    for why in skipped.values():
        out[why.split(":")[0]] += 1
    return dict(out)


# --------------------------------------------------------------------------
# 4. Patch -> changed symbols (ground truth for the retrieval benchmark)
# --------------------------------------------------------------------------


_HUNK_RE = re.compile(r"@@ -(\d+)(?:,(\d+))? \+\d+(?:,\d+)? @@")


def _patch_old_lines(patch: str) -> Dict[str, List[int]]:
    """Pre-patch line numbers that each file's hunks remove or replace.

    Coordinates are those of the *old* file, because the symbols are resolved
    against the pre-patch snapshot. Context lines are not edits and are skipped.
    A pure insertion is anchored on the old line it follows, so a line added
    inside a function body is attributed to that function. New files (``---
    /dev/null``) have no pre-patch symbols and are ignored.
    """
    per_file: Dict[str, List[int]] = defaultdict(list)
    lines = patch.splitlines()
    cur: Optional[str] = None
    in_hunk = False
    old_ln = 0
    i = 0
    while i < len(lines):
        line = lines[i]
        if line.startswith("--- ") and i + 1 < len(lines) and lines[i + 1].startswith("+++ "):
            path = line[4:].split("\t")[0].strip()
            cur = path[2:] if path.startswith("a/") else None
            in_hunk = False
            i += 2
            continue
        m = _HUNK_RE.match(line)
        if m:
            old_ln = int(m.group(1))
            if m.group(2) == "0":  # zero-length old range: insertion *after* line N
                old_ln += 1
            in_hunk = cur is not None
        elif in_hunk:
            if line.startswith("-"):
                per_file[cur].append(old_ln)
                old_ln += 1
            elif line.startswith("+"):
                per_file[cur].append(max(old_ln - 1, 1))
            elif line.startswith(" ") or line == "":
                old_ln += 1
            # "\ No newline at end of file" carries no line
        i += 1
    return per_file


def changed_files(patch: str) -> List[str]:
    """Pre-existing files the patch edits (new files excluded)."""
    return list(_patch_old_lines(patch))


def changed_symbols(patch: str, repo_dir: Path) -> List[Symbol]:
    """Symbols whose body the reference patch actually edits.

    Hunks are mapped back onto the enclosing class/function using the
    *pre-patch* snapshot source, which is the code the agent actually saw. Ids
    follow the same convention as ``enumerate_symbols`` (``mod.func``,
    ``mod.Class``, ``mod.Class.method``) so that they can be looked up in the
    graph. An edit inside a method counts against the method; the class is
    counted only for edits in its body outside any method. Edits inside nested
    functions count against the enclosing top-level symbol.
    """
    out: List[Symbol] = []
    for rel, lines in _patch_old_lines(patch).items():
        if not lines or _is_test_path(rel):
            continue
        path = repo_dir / rel
        if not path.exists():
            continue
        try:
            src = path.read_text(encoding="utf-8", errors="replace")
            tree = ast.parse(src)
        except (SyntaxError, ValueError, UnicodeDecodeError):
            continue
        mod = _module_name(rel)
        if not mod:
            continue
        for node in tree.body:
            lo, hi = _span(node) if hasattr(node, "lineno") else (0, -1)
            touched = {ln for ln in lines if lo <= ln <= hi}
            if not touched:
                continue
            if isinstance(node, ast.ClassDef):
                in_method: Set[int] = set()
                for sub in node.body:
                    if isinstance(sub, (ast.FunctionDef, ast.AsyncFunctionDef)):
                        slo, shi = _span(sub)
                        mine = {ln for ln in touched if slo <= ln <= shi}
                        if mine:
                            in_method |= mine
                            out.append(_func_symbol(f"{mod}.{node.name}.{sub.name}", sub, rel))
                if touched - in_method:
                    out.append(Symbol(f"{mod}.{node.name}", "class", False, rel, node.lineno, hi))
            elif isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                out.append(_func_symbol(f"{mod}.{node.name}", node, rel))
    # de-duplicate, keep order
    seen, uniq = set(), []
    for s in out:
        if s.node_id not in seen:
            seen.add(s.node_id)
            uniq.append(s)
    return uniq


# --------------------------------------------------------------------------
# 5. Retrieval benchmark - what does the corruption cost an agent?
# --------------------------------------------------------------------------


def _cosine_topk(emb: Dict[str, Any], queries: List["object"], k: int) -> List[List[str]]:
    import numpy as np

    ids, mat = _as_matrix(emb)
    if not ids or not queries:
        return [[] for _ in queries]
    q = np.stack([np.asarray(q, dtype=np.float32).reshape(-1) for q in queries])
    qn = np.linalg.norm(q, axis=1, keepdims=True)
    qn[qn == 0] = 1.0
    q = q / qn
    sims = q @ mat.T  # cosine, both sides L2-normalised
    kk = min(k, len(ids))
    idx = np.argpartition(-sims, kk - 1, axis=1)[:, :kk]
    return [[ids[j] for j in row[np.argsort(-sims[i, row])]] for i, row in enumerate(idx)]


def _neighbour_expansion(g: dict, seeds: Set[str], need: Set[str], hops: int = 2) -> Set[str]:
    """Symbols reachable from ``seeds`` by following ``calls`` edges.

    Models the multi-hop walk an agent performs with ``get_code_neighbors``,
    which follows edges in both directions.
    """
    nodes = set(graph_nodes(g))
    adj: Dict[str, Set[str]] = defaultdict(set)
    for e in g.get("edges", []):
        s, t = e.get("source"), e.get("target")
        if s in nodes and t in nodes:
            adj[s].add(t)
            adj[t].add(s)
    reached: Set[str] = set()
    frontier = {s for s in seeds if s in nodes}
    for _ in range(hops):
        nxt: Set[str] = set()
        for n in frontier:
            nxt |= adj.get(n, set())
        nxt -= reached
        reached |= nxt
        frontier = nxt
        if need & reached:
            break
    return reached


def retrieval_eval(
    root: str | Path, k_values: Sequence[int] = (1, 5, 10), limit: Optional[int] = None
) -> dict:
    """What does the corpus cost an agent, measured without an LLM?

    Ground truth = symbols whose bodies the reference ``patch`` edits, resolved
    against the pre-patch snapshot (the code the agent actually saw).

    Metrics, in order of how much they can be trusted:

    1. ``reachability`` - the fraction of needed symbols present in the node
       set at all. This is a HARD CEILING on ``search_similar_code``,
       ``get_code_neighbors`` and ``get_code_subgraph``: a symbol that is not
       a node can never be returned by any of the three, whatever the query.
       Reported overall and split by symbol kind.
    2. ``affected_tasks`` - how many tasks have at least one unreachable
       needed symbol, i.e. tasks an agent provably cannot complete with the
       graph tools.
    3. ``multi_hop`` - of the needed symbols that ARE present, how many can be
       reached by a 2-hop ``calls`` walk from a present symbol. This is what
       ``get_code_neighbors`` can actually exploit.
    4. ``self_retrieval_sanity`` - cosine recall@k when the query vector IS the
       target's own released vector. Reported only as an UPPER BOUND: it is
       self-retrieval, so it is near 1.0 by construction and says nothing about
       real text-query behaviour. The released embedder is not distributed, so
       a faithful text->vector benchmark is not reproducible offline.
    """
    root = Path(root)
    tasks = _load_tasks(root)
    if limit:
        tasks = tasks[:limit]

    per_task: List[dict] = []
    kind_tot = defaultdict(lambda: [0, 0])  # kind -> [present, total]
    kind_by_repo: Dict[str, Dict[str, List[int]]] = defaultdict(lambda: defaultdict(list))
    n_affected = 0
    n_with_async_need = 0
    n_async_unreachable_tasks = 0
    hop_hits = hop_total = 0
    micro = {k: [0, 0] for k in k_values}
    skipped: Dict[str, str] = {}

    with tempfile.TemporaryDirectory() as td:
        for i, t in enumerate(tasks, 1):
            _progress("bench", i, len(tasks))
            iid = t["instance_id"]
            tgz = root / "snapshots" / f"{iid}.tgz"
            gpath = root / "graphs" / f"{iid}.json"
            epath = root / "embeddings" / f"{iid}.npz"
            why = _skip_reason(tgz, gpath)
            if why:
                skipped[iid] = why
                continue
            repo_dir, why = _extract(tgz, td, iid)
            if repo_dir is None:
                skipped[iid] = why
                continue
            needed = changed_symbols(t.get("patch", ""), repo_dir)
            shutil.rmtree(Path(td) / iid, ignore_errors=True)
            if not needed:
                skipped[iid] = "no_needed_symbols" if t.get("patch") else "no_patch"
                continue

            g = load_graph(gpath)
            present = set(graph_nodes(g))
            emb: Dict[str, Any] = {}
            in_emb: Set[str] = set()
            if not _is_empty_file(epath):
                try:
                    emb = load_embeddings(epath)
                    in_emb = set(emb)
                except Exception:
                    emb = {}

            rec = {
                "instance_id": iid,
                "repo": t["repo"],
                "n_needed": len(needed),
                "n_needed_present": sum(1 for s in needed if s.node_id in present),
                "n_needed_embedded": sum(1 for s in needed if s.node_id in in_emb),
                "unreachable": [s.node_id for s in needed if s.node_id not in present][:10],
                "unreachable_kinds": sorted({s.kind for s in needed if s.node_id not in present}),
                "async_needed": sum(1 for s in needed if s.is_async),
                "async_present": sum(1 for s in needed if s.is_async and s.node_id in present),
            }
            rec["task_is_unreachable"] = rec["n_needed_present"] < len(needed)
            n_affected += int(rec["task_is_unreachable"])
            if rec["async_needed"]:
                n_with_async_need += 1
                if rec["async_present"] == 0:
                    n_async_unreachable_tasks += 1

            for s in needed:
                kind_tot[s.kind][0] += int(s.node_id in present)
                kind_tot[s.kind][1] += 1
                kind_by_repo[t["repo"]][s.kind].append(int(s.node_id in present))

            present_needed = {s.node_id for s in needed if s.node_id in present}
            other_present = present - present_needed
            if present_needed and other_present:
                reached = _neighbour_expansion(g, other_present, present_needed, hops=2)
                hit = len(present_needed & reached)
                hop_hits += hit
                hop_total += len(present_needed)
                rec["multi_hop_reachable"] = hit
                rec["multi_hop_total"] = len(present_needed)

            if emb and present_needed:
                vecs = [emb[n] for n in present_needed if n in emb]
                if vecs:
                    tops = _cosine_topk(emb, vecs, max(k_values))
                    for k in k_values:
                        hits = sum(1 for row in tops if present_needed & set(row[:k]))
                        micro[k][0] += hits
                        micro[k][1] += len(present_needed)
            per_task.append(rec)

    def pct(a: int, b: int) -> float:
        return round(100.0 * a / b, 2) if b else 0.0

    n_scored = len(per_task)
    total_present = sum(v[0] for v in kind_tot.values())
    total_needed = sum(v[1] for v in kind_tot.values())

    return {
        "n_tasks_scored": n_scored,
        "n_tasks_skipped": len(skipped),
        "skip_reasons": _count_reasons(skipped),
        "skipped": dict(list(skipped.items())[:20]),
        "reachability": {
            "needed_present": total_present,
            "needed_total": total_needed,
            "pct_needed_symbols_reachable": pct(total_present, total_needed),
            "by_kind": {
                k: {"present": v[0], "total": v[1], "pct": pct(v[0], v[1])} for k, v in kind_tot.items()
            },
            "by_repo": {
                r: {k: {"present": sum(v), "total": len(v), "pct": pct(sum(v), len(v))}
                    for k, v in kinds.items()}
                for r, kinds in kind_by_repo.items()
            },
        },
        "affected_tasks": {
            "n_tasks_with_unreachable_needed_symbol": n_affected,
            "pct_tasks_affected": pct(n_affected, n_scored),
            "n_tasks_needing_async_symbol": n_with_async_need,
            "n_tasks_fully_blind_to_async": n_async_unreachable_tasks,
        },
        "multi_hop": {
            "reached_within_2_hops": hop_hits,
            "present_needed_total": hop_total,
            "pct": pct(hop_hits, hop_total),
        },
        "self_retrieval_sanity_upper_bound": {
            "note": "query vector == target vector; near 1.0 by construction, NOT a text-query benchmark",
            **{
                f"recall@{k}": round(v[0] / v[1], 4) if v[1] else 0.0 for k, v in micro.items()
            },
        },
        "per_task": per_task,
    }

def reindex(
    root: str | Path,
    out_dir: str | Path,
    limit: Optional[int] = None,
    embed: bool = False,
) -> dict:
    """Emit corrected graphs with the missing (async) nodes inserted.

    Every symbol in the AST reference set that the graph lacks is added (on
    the released corpus these are overwhelmingly ``async def``). New nodes carry
    the released schema's three fields: the qualified ``id``, a matching
    ``name``, and the symbol's source ``text`` at ``base_commit``. No edges are
    synthesised and no vectors are written: the released embedder is not part
    of the dataset, so ``embed=True`` only records that backfill was skipped.
    """
    root = Path(root)
    out_dir = Path(out_dir)
    (out_dir / "graphs").mkdir(parents=True, exist_ok=True)

    tasks = _load_tasks(root)
    if limit:
        tasks = tasks[:limit]

    summary: dict = {"n_tasks": 0, "n_nodes_added": 0, "n_nodes_added_by_kind": defaultdict(int),
                     "per_task": [], "embedding_backfill": "skipped"}

    with tempfile.TemporaryDirectory() as td:
        for i, t in enumerate(tasks, 1):
            _progress("reindex", i, len(tasks))
            iid = t["instance_id"]
            gpath = root / "graphs" / f"{iid}.json"
            tgz = root / "snapshots" / f"{iid}.tgz"
            if not tgz.exists() or _is_empty_file(gpath):
                continue
            g = load_graph(gpath)
            present = set(graph_nodes(g))
            repo_dir, why = _extract(tgz, td, iid)
            if repo_dir is None:
                summary.setdefault("skipped", {})[iid] = why
                continue

            added: List[dict] = []
            sources: Dict[str, List[str]] = {}
            for s in enumerate_symbols(repo_dir):
                if s.node_id in present:
                    continue
                # the reference set includes symbols outside the repo package
                # root; only backfill what the module path can justify
                if not s.node_id.split(".")[0]:
                    continue
                if s.file not in sources:
                    sources[s.file] = (repo_dir / s.file).read_text(
                        encoding="utf-8", errors="replace").splitlines()
                text = "\n".join(sources[s.file][s.lineno - 1 : s.end_lineno or s.lineno])
                added.append({"id": s.node_id, "name": s.node_id, "text": text, "kind": s.kind,
                              "file": s.file, "lineno": s.lineno})
            shutil.rmtree(Path(td) / iid, ignore_errors=True)

            if not added:
                continue
            g["nodes"].extend(
                {"id": a["id"], "name": a["name"], "text": a["text"]} for a in added
            )
            if not any(k == "cgr_backfilled" for k in g.get("graph", {})):
                g.setdefault("graph", {})["cgr_backfilled"] = len(added)
            dest = out_dir / "graphs" / f"{iid}.json"
            dest.write_text(json.dumps(g), encoding="utf-8")

            summary["n_tasks"] += 1
            summary["n_nodes_added"] += len(added)
            for a in added:
                summary["n_nodes_added_by_kind"][a["kind"]] += 1
            summary["per_task"].append(
                {
                    "instance_id": iid,
                    "repo": t["repo"],
                    "n_added": len(added),
                    "n_added_async": sum(1 for a in added if a["kind"] == "async_function"),
                    "examples": [a["id"] for a in added[:5]],
                }
            )
    summary["n_nodes_added_by_kind"] = dict(summary["n_nodes_added_by_kind"])
    if embed:
        summary["embedding_backfill"] = (
            "requested but not performed: the released embedder is not distributed "
            "with the dataset, so new nodes ship without vectors"
        )
    return summary

def unpack_zip(zip_path: str | Path, dest: str | Path) -> Path:
    dest = Path(dest)
    dest.mkdir(parents=True, exist_ok=True)
    with zipfile.ZipFile(zip_path) as zf:
        for m in zf.infolist():
            name = m.filename
            if name.startswith("/") or ".." in Path(name).parts:
                raise ValueError(f"unsafe path in archive: {name}")
        zf.extractall(dest)
    # Kaggle archives often nest everything one level deep
    entries = [p for p in dest.iterdir() if not p.name.startswith(".")]
    if len(entries) == 1 and entries[0].is_dir():
        root = entries[0]
        if (root / _TASK_FILE).exists() or (root / "graphs").is_dir():
            return root
    return dest


def find_root(start: str | Path) -> Path:
    """Locate the directory that actually holds tasks.jsonl."""
    start = Path(start)
    if (start / _TASK_FILE).exists():
        return start
    for p in start.rglob(_TASK_FILE):
        return p.parent
    raise FileNotFoundError(f"no {_TASK_FILE} found under {start}")

def apply_repaired(root: str | Path, repaired_dir: str | Path) -> dict:
    """Install re-indexed graphs over the released ones, so that a second
    ``audit()`` / ``retrieval_eval()`` pass measures the repaired corpus.

    Backs up nothing - the released files are still on Kaggle - but writes a
    manifest of what changed.
    """
    root = Path(root)
    src = Path(repaired_dir) / "graphs"
    if not src.is_dir():
        raise FileNotFoundError(f"no graphs in {src}")
    changed = []
    for p in sorted(src.glob("*.json")):
        dest = root / "graphs" / p.name
        before = _sha256(dest) if dest.exists() else None
        shutil.copyfile(p, dest)
        after = _sha256(dest)
        if before != after:
            changed.append(p.name)
    return {
        "n_installed": len(list(src.glob("*.json"))),
        "n_changed": len(changed),
        "changed": changed[:50],
    }


def structural_scan(root: str | Path) -> dict:
    """Secondary finding: how much of each graph is actually a graph.

    Reports edge-type diversity, edge density, module-level node count and the
    per-repository node budget. Read-only.
    """
    root = Path(root)
    edge_types: Set[str] = set()
    mod_nodes = n_nodes_total = n_edges_total = 0
    per_repo: Dict[str, int] = {}
    n_tasks = 0
    for t in _load_tasks(root):
        g = root / "graphs" / f"{t['instance_id']}.json"
        if _is_empty_file(g):
            continue
        n_tasks += 1
        d = load_graph(g)
        nodes = graph_nodes(d)
        edge_types |= {e.get("type") for e in d.get("edges", []) if e.get("type")}
        n_nodes_total += len(nodes)
        n_edges_total += len(d.get("edges", []))
        mod_nodes += sum(1 for x in nodes if x.count(".") <= 1)
        per_repo[t["repo"]] = per_repo.get(t["repo"], 0) + len(nodes)
    return {
        "n_tasks": n_tasks,
        "distinct_edge_types": sorted(edge_types),
        "total_nodes": n_nodes_total,
        "total_edges": n_edges_total,
        "mean_edges_per_node": round(n_edges_total / max(n_nodes_total, 1), 3),
        "module_level_nodes": mod_nodes,
        "nodes_by_repo": per_repo,
    }


def full_report(root: str | Path, repaired_dir: Optional[str | Path] = None,
                k_values: Sequence[int] = (1, 5, 10),
                limit: Optional[int] = None, embed: bool = False,
                dry_run: bool = False) -> dict:
    """Run the whole pipeline and return the canonical `results.json` schema.

    This is the single definition of that schema; the notebook builds the same
    keys step by step so a reader can see each stage, and `cgr.py report` calls
    this function. The two therefore cannot drift silently.
    """
    root = find_root(root)
    repaired_dir = Path(repaired_dir) if repaired_dir else root.parent / "repaired"

    c0 = census(root)
    rep = repair(root, dry_run=dry_run)
    c1 = census(root)
    audit_before = audit(root, limit=limit)
    bench_before = retrieval_eval(root, k_values=tuple(k_values), limit=limit)
    rx = reindex(root, repaired_dir, limit=limit, embed=embed)
    inst = apply_repaired(root, repaired_dir)
    audit_after = audit(root, limit=limit)
    bench_after = retrieval_eval(root, k_values=tuple(k_values), limit=limit)
    structural = structural_scan(root)

    return {
        "tool": "cgr",
        "schema": 1,
        "corpus": {"n_tasks": c0["n_tasks"], "repos": c0["repos"]},
        "census_before": c0,
        "repair": rep,
        "census_after": c1,
        "audit_before": audit_before,
        "audit_after": audit_after,
        "bench_before": bench_before,
        "bench_after": bench_after,
        "reindex": {k: v for k, v in rx.items() if k != "per_task"},
        "install": inst,
        "structural": structural,
    }


def _emit(obj: Any, out: Optional[str]) -> None:
    text = json.dumps(obj, indent=2, default=str)
    if out:
        Path(out).parent.mkdir(parents=True, exist_ok=True)
        Path(out).write_text(text, encoding="utf-8")
        print(f"wrote {out} ({len(text):,} bytes)")
    else:
        print(text)


def main(argv: Optional[Sequence[str]] = None) -> int:
    ap = argparse.ArgumentParser("cgr", description="Code Graph Repair for the Gemma 4 Developer Agent corpus")
    ap.add_argument("command", choices=["census", "repair", "audit", "reindex", "bench", "report",
                                       "unpack", "apply"])
    ap.add_argument("--root", required=True, help="dataset root containing tasks.jsonl")
    ap.add_argument("--zip", help="(unpack) path to a Kaggle-downloaded .zip")
    ap.add_argument("--out", help="where to write the JSON result (stdout if omitted)")
    ap.add_argument("--out-dir", help="(reindex) directory to write corrected graphs into")
    ap.add_argument("--dry-run", action="store_true")
    ap.add_argument("--limit", type=int, default=None, help="only the first N tasks (fast smoke run)")
    ap.add_argument("--k", type=int, nargs="+", default=[1, 5, 10])
    ap.add_argument("--embed", action="store_true")
    args = ap.parse_args(argv)

    if args.command == "unpack":
        dest = args.out_dir or args.out or "data"
        root = unpack_zip(args.zip, dest)
        print(root)
        return 0

    root = find_root(args.root)

    if args.command == "census":
        _emit(census(root), args.out)
    elif args.command == "repair":
        _emit(repair(root, dry_run=args.dry_run), args.out)
    elif args.command == "audit":
        _emit(audit(root, limit=args.limit), args.out)
    elif args.command == "reindex":
        dest = args.out_dir or "repaired"
        _emit(reindex(root, dest, limit=args.limit, embed=args.embed), args.out)
    elif args.command == "bench":
        _emit(retrieval_eval(root, k_values=tuple(args.k), limit=args.limit), args.out)
    elif args.command == "apply":
        _emit(apply_repaired(root, args.out_dir or "repaired"), args.out)
    elif args.command == "report":
        print("note: 'report' repairs and re-indexes IN PLACE; run it on a copy.", file=sys.stderr)
        _emit(full_report(root, repaired_dir=args.out_dir, k_values=tuple(args.k),
                          limit=args.limit, embed=args.embed, dry_run=args.dry_run), args.out)
    return 0

    if not cred.exists() and not (WORK / "kaggle.json").exists():
        raise SystemExit(
            "No attached dataset and no ~/.kaggle/kaggle.json.\n"
            "Either add the competition as a notebook input, or upload your "
            f"kaggle.json to {WORK}/ and re-run this cell."
        )
    if not shutil.which("kaggle"):
        pip_install("kaggle")
    # The release is ~22.4 GB compressed and ~45 GB once unpacked.
    free = shutil.disk_usage(WORK).free / 1e9
    if free < 50:
        print(f"WARNING: {free:.1f} GB free, need ~50 GB. In Colab pick a larger disk "
              "under Runtime > Change runtime type.")
    zip_path = WORK / f"{COMP}.zip"
    if not zip_path.exists():
        subprocess.run(["kaggle", "competitions", "download", "-c", COMP, "-p", str(WORK)], check=True)
    dest = WORK / "corpus"
    print("extracting ~22 GB, this takes a while ...")
    import zipfile
    with zipfile.ZipFile(zip_path) as z:
        z.extractall(dest)
    for p in dest.rglob("tasks.jsonl"):
        if p.parent.joinpath("graphs").is_dir():
            return p.parent
    raise SystemExit("tasks.jsonl not found after extraction")

def find_snapshots(root):
    # Normally snapshots/ sits next to tasks.jsonl; otherwise search the input tree.
    if (root / "snapshots").is_dir():
        return root / "snapshots"
    for base in (root.parent, root.parent.parent, Path("/kaggle/input")):
        if base.is_dir():
            for p in base.rglob("snapshots"):
                if p.is_dir() and any(p.glob("*.tgz")):
                    return p
    return None

def ensure_writable(root):
    # cgr repairs in place, but /kaggle/input is read-only. Stage a fresh
    # writable copy of the parts cgr mutates (tasks.jsonl, graphs/, embeddings/)
    # and link the large, read-only snapshots/ instead of copying them.
    probe = root / ".cgr_write_probe"
    try:
        probe.write_text("x")
        probe.unlink()
        return root
    except OSError:
        pass
    dest = WORK / "corpus_rw"
    if dest.exists():
        shutil.rmtree(dest)  # always start from the released state
    dest.mkdir(parents=True)
    # copyfile, not copy2/copytree: those carry over the read-only file mode
    shutil.copyfile(root / "tasks.jsonl", dest / "tasks.jsonl")
    for sub in ("graphs", "embeddings"):
        if (root / sub).is_dir():
            (dest / sub).mkdir()
            for f in (root / sub).iterdir():
                if f.is_file():
                    shutil.copyfile(f, dest / sub / f.name)
    snaps = find_snapshots(root)
    if snaps:
        (dest / "snapshots").symlink_to(snaps, target_is_directory=True)
        print("snapshots:", snaps)
    else:
        print("WARNING: no snapshots/ directory with .tgz files found; the audit will score nothing")
    size = sum(f.stat().st_size for f in dest.rglob("*") if f.is_file() and not f.is_symlink())
    print(f"input is read-only: staged a writable copy at {dest} ({size / 1e9:.2f} GB)")
    return dest
