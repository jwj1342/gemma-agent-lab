# CELL 4 code
VALIDATOR = "\"\"\"Local pre-flight validator mirroring the rules in HARNESS_README.md (sections 2-3, 8).\"\"\"\nimport sys, os, zipfile, yaml, json, re\n\nALLOWED_MODEL = \"gemma-4-31b-it-qat-w4a16-ct\"\nEXT = {\".yaml\", \".yml\", \".md\", \".txt\", \".py\", \".json\", \".safetensors\"}\nTOOLS = {\"run_command\",\"submit_patch\",\"get_status\",\"read_file\",\"edit_file\",\"write_file\",\n         \"get_code_neighbors\",\"search_similar_code\",\"get_code_subgraph\"}\nROOTS = [\"agent.yaml\",\"agent.yml\",\"root_agent.yaml\",\"root_agent.yml\"]\nMAX_SIZE = 3*1024**3\nGEN_OK = {\"temperature\",\"top_p\",\"top_k\",\"max_output_tokens\",\"presence_penalty\",\"frequency_penalty\",\n          \"stop_sequences\",\"response_mime_type\",\"seed\",\"thinking_config\"}\n\nclass Loader(yaml.SafeLoader): pass\ndef _include(loader, node, base=[None]):\n    rel = loader.construct_scalar(node)\n    root = loader.root_dir\n    if os.path.isabs(rel) or \"..\" in rel.split(\"/\") or \"\\0\" in rel:\n        raise ValueError(f\"PathTraversal: {rel}\")\n    full = os.path.realpath(os.path.join(os.path.dirname(loader.name), rel))\n    if not full.startswith(os.path.realpath(root)): raise ValueError(f\"escapes root: {rel}\")\n    if full.endswith((\".md\",\".txt\")): return open(full, encoding=\"utf-8\").read()\n    return load_yaml(full, root, loader.depth+1)\nLoader.add_constructor(\"!include\", _include)\n\ndef load_yaml(path, root, depth=0):\n    if depth > 10: raise ValueError(\"include depth > 10\")\n    with open(path, encoding=\"utf-8\") as f:\n        l = Loader(f); l.root_dir = root; l.depth = depth\n        try: return l.get_single_data()\n        finally: l.dispose()\n\ndef check(d):\n    errs, models = [], set()\n    root = os.path.realpath(d)\n    files = [os.path.join(r,f) for r,_,fs in os.walk(d) for f in fs]\n    if len(files) > 10000: errs.append(\"too many files\")\n    size = sum(os.path.getsize(f) for f in files)\n    if size >= MAX_SIZE: errs.append(\"size >= 3 GiB\")\n    for f in files:\n        if os.path.islink(f): errs.append(f\"symlink: {f}\")\n        if os.path.splitext(f)[1].lower() not in EXT: errs.append(f\"disallowed extension: {f}\")\n    found = [r for r in ROOTS if os.path.exists(os.path.join(d,r))]\n    if len(found) != 1: errs.append(f\"need exactly one root config, found {found}\"); return errs, size\n    seen = set()\n    def walk(path, depth=0):\n        if depth > 50 or path in seen: return\n        seen.add(path)\n        cfg = load_yaml(path, root)\n        cls = cfg.get(\"agent_class\",\"LlmAgent\")\n        if cls == \"LlmAgent\":\n            m = cfg.get(\"model\")\n            if not m: errs.append(f\"{path}: model missing\")\n            else: models.add(re.sub(r\"^(openai|google|hosted_vllm|custom)/\",\"\",m))\n            if \"{problem_description}\" not in str(cfg.get(\"instruction\",\"\")) and depth==0:\n                errs.append(\"root instruction does not use {problem_description}\")\n            for t in cfg.get(\"tools\",[]):\n                if isinstance(t,str):\n                    if t not in TOOLS: errs.append(f\"{path}: unknown tool {t}\")\n                elif \"agent_tool\" in t:\n                    walk(os.path.join(d,t[\"agent_tool\"][\"config_path\"]), depth+1)\n            for s in cfg.get(\"sub_agents\",[]): walk(os.path.join(d,s[\"config_path\"]), depth+1)\n            g = cfg.get(\"generate_content_config\") or {}\n            for k in g:\n                if k not in GEN_OK: errs.append(f\"{path}: forbidden generation field {k}\")\n            if g.get(\"max_output_tokens\",1) > 32768: errs.append(\"max_output_tokens > 32768\")\n            tb = (g.get(\"thinking_config\") or {}).get(\"thinking_budget\",0)\n            if tb > 32768: errs.append(\"thinking_budget > 32768\")\n            if \"thinking_level\" in (g.get(\"thinking_config\") or {}):\n                errs.append(f\"{path}: omit thinking_level on vLLM; use include_thoughts+thinking_budget\")\n            if g.get(\"max_output_tokens\",16384) + tb > 32768: errs.append(\"output+thinking exceeds 32768 context\")\n        else:\n            for s in cfg.get(\"sub_agents\",[]): walk(os.path.join(d,s[\"config_path\"]), depth+1)\n    walk(os.path.join(d, found[0]))\n    if models != {ALLOWED_MODEL}: errs.append(f\"models must be exactly {{{ALLOWED_MODEL}}}, got {models}\")\n    return errs, size\n\ndef check_zip(z):\n    names = zipfile.ZipFile(z).namelist()\n    return [] if any(n in ROOTS for n in names) else [\"root config must be at the ZIP ROOT (no wrapper folder)\"]\n\ndef check_parquet(p, tasks_jsonl):\n    import pandas as pd\n    df = pd.read_parquet(p); errs = []\n    if list(df.columns) != [\"id\",\"prediction\"]: errs.append(f\"columns must be [id, prediction], got {list(df.columns)}\")\n    ids = {json.loads(l)[\"instance_id\"] for l in open(tasks_jsonl)}\n    if set(df[\"id\"]) != ids: errs.append(f\"id mismatch: missing {len(ids-set(df['id']))}, extra {len(set(df['id'])-ids)}\")\n    for i,r in df.iterrows():\n        v = r[\"prediction\"]\n        if v != \"NO_PATCH\" and not (v.startswith(\"diff --git\") or v.startswith(\"--- \")): errs.append(f\"{r['id']}: not a unified diff\")\n        if v != \"NO_PATCH\" and re.search(r\"^(\\+\\+\\+|---) .*(test_|/tests?/|conftest|pytest\\.ini)\", v, re.M):\n            errs.append(f\"{r['id']}: touches tests/config (discarded by harness)\")\n    return errs\n\nif __name__ == \"__main__\":\n    d = sys.argv[1]\n    errs, size = check(d)\n    if len(sys.argv) > 2: errs += check_zip(sys.argv[2])\n    print(f\"size={size} bytes\"); print(\"OK\" if not errs else \"\\n\".join(\"ERROR: \"+e for e in errs))\n    sys.exit(1 if errs else 0)\n"
